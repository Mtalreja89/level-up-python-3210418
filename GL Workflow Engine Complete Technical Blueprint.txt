 SECTION 1: DATABASE ARCHITECTURE & TABLE DOCUMENTATION
--------------------------------------------------------------------------------

The database schema is organized into four distinct architectural layers:

1. REFERENCE LAYER (Master Data)
   - DATA_SOURCES: Master list of upstream/downstream financial systems (e.g., FXO, POWER, GECD).
   - COMPONENT_TYPES: Master list of allowed workflow block types (data_transfer, exclusion_type, filter_type, control_type, pipeline, logical_component).
   - DATA_SOURCE_COMPONENTS: Mapping between component types and data sources. NULL data_source_id represents a Global Logical Component.
   - DATA_SOURCE_COMPONENT_TASKS: Reusable task function catalog (pre, main, post phases) available for components.

2. PIPELINE MASTER & SCHEDULE LAYER
   - PIPELINES: Top-level pipeline definitions tied to a specific data source, carrying business justification and top-level spec links.
   - PIPELINE_SCHEDULES: Strict 1:1 relationship with pipelines storing cron expressions, execution intervals, and active schedule flags.

3. RUNTIME PIPELINE INSTANCE & EDGE LAYER
   - PIPELINE_COMPONENTS: Instances of components dragged onto the canvas for a specific pipeline, tracking sequence numbers and child/nested pipelines.
   - PIPELINE_COMPONENT_EDGES: Inter-component conditional branching routes (success, failure, custom expression evaluation).
   - COMPONENT_TASK_EDGES: Intra-component task-level failure recovery and branching routes.

4. DATA TRANSFER & FIELD TRANSFORMATIONS LAYER
   - TASK_TABLE_MAPPINGS: Direct source-to-destination table pairs anchored to the Main phase task of a data_transfer component.
   - TASK_FIELD_TRANSFORM_MAPPINGS: Column-level mapping rules (Column A -> Column B) with transformation functions, parameters, and audit justifications.

-- 1. DATA_SOURCES
CREATE TABLE data_sources (
    data_source_id NUMBER(10) NOT NULL,
    data_source_name VARCHAR2(50) NOT NULL,
    description VARCHAR2(250),
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT pk_data_sources PRIMARY KEY (data_source_id),
    CONSTRAINT uk_data_source_name UNIQUE (data_source_name),
    CONSTRAINT chk_ds_active CHECK (is_active IN ('Y', 'N'))
);

-- 2. COMPONENT_TYPES
CREATE TABLE component_types (
    component_type_id NUMBER(10) NOT NULL,
    component_type_name VARCHAR2(50) NOT NULL,
    description VARCHAR2(250),
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT pk_component_types PRIMARY KEY (component_type_id),
    CONSTRAINT uk_component_type_name UNIQUE (component_type_name),
    CONSTRAINT chk_ct_active CHECK (is_active IN ('Y', 'N'))
);

-- 3. DATA_SOURCE_COMPONENTS
CREATE TABLE data_source_components (
    ds_component_id NUMBER(10) NOT NULL,
    data_source_id NUMBER(10) NULL, -- NULL indicates global logical component
    component_type_id NUMBER(10) NOT NULL,
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT pk_data_source_components PRIMARY KEY (ds_component_id),
    CONSTRAINT fk_dsc_datasource FOREIGN KEY (data_source_id) REFERENCES data_sources(data_source_id),
    CONSTRAINT fk_dsc_componenttype FOREIGN KEY (component_type_id) REFERENCES component_types(component_type_id),
    CONSTRAINT chk_dsc_active CHECK (is_active IN ('Y', 'N'))
);

-- 4. DATA_SOURCE_COMPONENT_TASKS
CREATE TABLE data_source_component_tasks (
    task_id NUMBER(10) NOT NULL,
    ds_component_id NUMBER(10) NOT NULL,
    phase VARCHAR2(20) NOT NULL,
    task_name VARCHAR2(150) NOT NULL,
    description VARCHAR2(250),
    sequence_number NUMBER(10) NOT NULL,
    priority NUMBER(10) DEFAULT 0 NOT NULL,
    run_in_parallel CHAR(1) DEFAULT 'N' NOT NULL,
    trigger_rule VARCHAR2(50) DEFAULT 'all_success' NOT NULL,
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    functional_spec_url VARCHAR2(500),
    technical_spec_url VARCHAR2(500),
    task_docstring CLOB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT pk_ds_component_tasks PRIMARY KEY (task_id),
    CONSTRAINT fk_dsct_ds_component FOREIGN KEY (ds_component_id) REFERENCES data_source_components(ds_component_id),
    CONSTRAINT chk_dsct_phase CHECK (phase IN ('pre', 'main', 'post')),
    CONSTRAINT chk_dsct_parallel CHECK (run_in_parallel IN ('Y', 'N')),
    CONSTRAINT chk_dsct_active CHECK (is_active IN ('Y', 'N'))
);

-- 5. PIPELINES
CREATE TABLE pipelines (
    pipeline_id NUMBER(10) NOT NULL,
    pipeline_name VARCHAR2(150) NOT NULL,
    data_source_id NUMBER(10) NOT NULL,
    description VARCHAR2(500),
    is_active CHAR(1) DEFAULT 'Y' NOT NULL,
    functional_spec_url VARCHAR2(500),
    technical_spec_url VARCHAR2(500),
    business_justification CLOB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT pk_pipelines PRIMARY KEY (pipeline_id),
    CONSTRAINT fk_pipeline_datasource FOREIGN KEY (data_source_id) REFERENCES data_sources(data_source_id),
    CONSTRAINT chk_pipeline_active CHECK (is_active IN ('Y', 'N'))
);

-- 6. PIPELINE_SCHEDULES
CREATE TABLE pipeline_schedules (

    schedule_type VARCHAR2(30) NOT NULL,

    interval_seconds NUMBER(10),
    timezone VARCHAR2(50) DEFAULT 'UTC' NOT NULL,

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { DragDropModule, CdkDragDrop, moveItemInArray } from '@angular/cdk/drag-and-drop';
import { DataSource, ComponentType, PipelineComponentInstance, PipelinePayload } from './workflow.model';
import { WorkflowService } from './workflow.service';

@Component({
  selector: 'app-workflow-builder',
  standalone: true,
  imports: [CommonModule, FormsModule, DragDropModule],
  templateUrl: './workflow-builder.component.html',
  styleUrls: ['./workflow-builder.component.scss']
})
export class WorkflowBuilderComponent implements OnInit {
  dataSources: DataSource[] = [];
  availableComponentTypes: ComponentType[] = [];

  selectedDataSourceId: number | null = null;
  pipelineName: string = '';
  pipelineDescription: string = '';
  functionalSpecUrl: string = '';
  technicalSpecUrl: string = '';
  cronExpression: string = '0 5 * * *';

  canvasComponents: PipelineComponentInstance[] = [];
  selectedComponent: PipelineComponentInstance | null = null;

  constructor(private workflowService: WorkflowService) {}

  ngOnInit(): void {
    this.workflowService.getDataSources().subscribe(res => this.dataSources = res);
  }

  onDataSourceChange(dataSourceId: number): void {
    this.selectedDataSourceId = dataSourceId;
    // Fetches DS-specific components AND global Logical Components (data_source_id IS NULL)
    this.workflowService.getComponentsForDataSource(dataSourceId).subscribe(res => {
      this.availableComponentTypes = res;
    });
  }

  onDrop(event: CdkDragDrop<any[]>): void {
    if (event.previousContainer === event.container) {
      moveItemInArray(this.canvasComponents, event.previousIndex, event.currentIndex);
      this.resequence();
    } else {
      const compType: ComponentType = event.previousContainer.data[event.previousIndex];
      const newInstance: PipelineComponentInstance = {
        componentInstanceId: 'comp_' + Math.random().toString(36).substring(2, 9),
        dsComponentId: compType.componentTypeId,
        componentTypeName: compType.componentTypeName,
        componentAlias: `${compType.componentTypeName}_${this.canvasComponents.length + 1}`,
        sequenceNumber: event.currentIndex + 1,
        nestedPipelineId: compType.componentTypeName === 'pipeline' ? null : undefined,
        tableMappings: compType.componentTypeName === 'data_transfer' ? [] : undefined
      };
      this.canvasComponents.splice(event.currentIndex, 0, newInstance);
      this.resequence();
      this.selectComponent(newInstance);
    }
  }

  private resequence(): void {
    this.canvasComponents.forEach((comp, idx) => comp.sequenceNumber = idx + 1);
  }

  selectComponent(comp: PipelineComponentInstance): void {
    this.selectedComponent = comp;
  }

  savePipeline(): void {
    if (!this.selectedDataSourceId || !this.pipelineName) {
      alert('Please fill required pipeline headers.');
      return;
    }
    const payload: PipelinePayload = {
      pipelineName: this.pipelineName,
      dataSourceId: this.selectedDataSourceId,
      description: this.pipelineDescription,
      functionalSpecUrl: this.functionalSpecUrl,
      technicalSpecUrl: this.technicalSpecUrl,
      schedule: {
        scheduleType: 'cron',
        cronExpression: this.cronExpression,
        timezone: 'UTC'
      },
      components: this.canvasComponents
    };
    this.workflowService.savePipeline(payload).subscribe(() => alert('Saved successfully!'));
  }
}
